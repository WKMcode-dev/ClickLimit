cmake_minimum_required(VERSION 3.20)
project(ClickLimit VERSION 0.7.1 LANGUAGES CXX RC)
include(CTest)
if(NOT WIN32)
  message(FATAL_ERROR "ClickLimit requer Windows")
endif()
add_executable(ClickLimit WIN32 src/main.cpp src/application.rc)
set_property(SOURCE src/application.rc PROPERTY INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/src")
include(CheckIPOSupported)
check_ipo_supported(RESULT ipo_supported OUTPUT ipo_error)
if(ipo_supported)
  set_property(TARGET ClickLimit PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endif()
target_compile_features(ClickLimit PRIVATE cxx_std_20)
target_compile_definitions(ClickLimit PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_link_libraries(ClickLimit PRIVATE user32 gdi32 dwmapi uxtheme shell32 comctl32)
add_executable(ClickLimitBenchmark src/benchmark.cpp)
target_compile_features(ClickLimitBenchmark PRIVATE cxx_std_20)
target_compile_definitions(ClickLimitBenchmark PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_link_libraries(ClickLimitBenchmark PRIVATE user32)
if(ipo_supported)
  set_property(TARGET ClickLimitBenchmark PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endif()
if(BUILD_TESTING)
  add_executable(ClickLimitTests tests/engine_tests.cpp)
  target_include_directories(ClickLimitTests PRIVATE src)
  target_compile_features(ClickLimitTests PRIVATE cxx_std_20)
  add_test(NAME engine_and_layout COMMAND ClickLimitTests)
  add_test(NAME hidden_ui_resize_scroll COMMAND ClickLimit --ui-test "${CMAKE_CURRENT_BINARY_DIR}/ui-regression.txt")
  add_test(NAME simulated_acceptance_benchmark COMMAND ClickLimitBenchmark --simulate "${CMAKE_CURRENT_BINARY_DIR}/simulated-benchmark.csv")
endif()
if(MSVC)
  set_property(TARGET ClickLimit PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(ClickLimit PRIVATE /W4 /permissive- /utf-8)
  set_property(TARGET ClickLimitBenchmark PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(ClickLimitBenchmark PRIVATE /W4 /permissive- /utf-8)
endif()
