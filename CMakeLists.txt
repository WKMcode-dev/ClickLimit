cmake_minimum_required(VERSION 3.20)
project(ClickLimit LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "ClickLimit requer Windows")
endif()
add_executable(ClickLimit src/main.cpp)
target_compile_features(ClickLimit PRIVATE cxx_std_20)
target_compile_definitions(ClickLimit PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(ClickLimit PRIVATE user32)
if(MSVC)
  target_compile_options(ClickLimit PRIVATE /W4 /permissive- /utf-8)
endif()
